sem nueva_medicion;
sem mutex_driver;
int esperando;

int driver_init(){
    nueva_medicion.sem(0);
    mutex_driver.sem(1);
    esperando = 0;
    request_irq(IRQ_SENSE_FINISHED, &handler_nueva_medicion);

    return IO_OK
}

void handler_nueva_medicion(){
    mutex_driver.wait();
        if (esperando > 0){
            esperando --;
            nueva_medicion.signal();
        }
    mutex_driver.signal();
}

int driver_remove(){
    nueva_medicion.destroy();
    mutex_driver.destroy();
    free_irq(IRQ_SENSE_FINISHED);

    return IO_OK
}
