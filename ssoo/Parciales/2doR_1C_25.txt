int procesos_activos[5];
semaphore mutex_open;
semaphore mutex_write;

int driver_init(){
    mutex_open.sem(1);
    mutex_write.sem(1);
}

int driver_remove(){
    mutex_open.destroy();
    mutex_write.destroy();
}

int driver_open(pid_t pid) {
    sema_wait(&mutex_open); // Protección para concurrencia

    // 1. Verificar si el proceso ya estaba registrado previa/abierto
    for (int i = 0; i < MAX_PROCESOS; i++) {
        if (procesos_activos[i] == pid) {
            sema_signal(&mutex_open);
            return IO_OK; // Ya estaba activo
        }
    }

    // 2. Buscar una ranura libre para el nuevo proceso
    for (int i = 0; i < MAX_PROCESOS; i++) {
        if (procesos_activos[i] == 0) { // 0 representa libre
            procesos_activos[i] = pid;

            // Aquí se asignaría/notificaría el hardware si corresponde
            
            sema_signal(&mutex_open);
            return IO_OK; // Conexión exitosa -> Devuelve IO_OK al Administrador de E/S
        }
    }

    // 3. Si no hay vacantes (ya hay 5 procesos), se falla[cite: 2]
    sema_signal(&mutex_open);
    return IO_ERROR; // Devuelve error -> el SO le devolverá -1/error al usuario
}

int driver_close(pid_t pid) {
    sema_wait(&mutex_open);

    for (int i = 0; i < MAX_PROCESOS; i++) {
        if (procesos_activos[i] == pid) {
            procesos_activos[i] = 0; // Liberar ranura

            // Aquí se apagarían recursos/LEDs asociados a este PID si aplica

            sema_signal(&mutex_open);
            return IO_OK; // Desconexión exitosa
        }
    }

    sema_signal(&mutex_open);
    return IO_ERROR; // El proceso no estaba registrado
}
